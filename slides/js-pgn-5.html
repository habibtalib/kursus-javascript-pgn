const JEJARI_BUMI_KM = 6371;
const rad = (d) => (d * Math.PI) / 180;

export function jarakKm([lng1, lat1], [lng2, lat2]) {
  const dLat = rad(lat2 - lat1);
  const dLng = rad(lng2 - lng1);
  const a =
    Math.sin(dLat / 2) ** 2 +
    Math.cos(rad(lat1)) * Math.cos(rad(lat2)) *
    Math.sin(dLng / 2) ** 2;
  return 2 * JEJARI_BUMI_KM * Math.asin(Math.sqrt(a));
}

export function bboxDari(features) {
  const b = [Infinity, Infinity, -Infinity, -Infinity];
  for (const { geometry: { coordinates: [lng, lat] } } of features) {

const BASE = 'http://localhost:3000/api';
const KUNCI_API = 'latihan-pgn-2026';   // ⚠️ training mock only — not a real secret
export class ApiError extends Error {
  constructor(status, badan) {
    super(badan?.ralat ?? `HTTP ${status}`);
    this.name = 'ApiError';
    this.status = status;
    this.medan = badan?.medan ?? {};     // 422 validation error
  }
}

export async function mintaJson(laluan, { method = 'GET', body, signal, timeoutMs = 8000 } = {}) {
  const headers = { Accept: 'application/json' };
  if (body !== undefined) headers['Content-Type'] = 'application/json';
  if (method !== 'GET') headers['X-API-Key'] = KUNCI_API;
  const tamat = AbortSignal.timeout(timeoutMs);
  const res = await fetch(`${BASE}${laluan}`, {
    method, headers,
    body: body === undefined ? undefined : JSON.stringify(body),
    signal: signal ? AbortSignal.any([signal, tamat]) : tamat,
  });
  if (res.status === 204) return null;              // DELETE: no body
  const data = await res.json().catch(() => null);  // body might not be JSON
  if (!res.ok) throw new ApiError(res.status, data);
  return data;
}

}

export function ciptaStore(awal) {
  let state = structuredClone(awal);
  const pendengar = new Set();

  return {
    dapat: () => state,

    },
  };

const tidur = (ms) => new Promise((r) => setTimeout(r, ms));

export async function denganRetry(fn, cubaan = 3, asas = 400) {
  for (let i = 1; ; i++) {
    try {
      return await fn();
    } catch (err) {
      const ulang = !(err instanceof ApiError) || err.status >= 500;
      if (!ulang || i >= cubaan) throw err;
      await tidur(asas * 2 ** (i - 1));   // 400, 800 ms…
    }
  }
}

import { test } from 'node:test';
import assert from 'node:assert/strict';
import {
  jarakKm, dalamMalaysia, kiraIkut, formatKoordinat,

test('dalamMalaysia menolak koordinat terbalik', () => {
  assert.equal(dalamMalaysia([101.6958, 2.9264]), true);

test('jarakKm Putrajaya → KLCC ≈ 25.8 km', () => {
  const d = jarakKm([101.6958, 2.9264], [101.7117, 3.1579]);

    { baharu: 2, selesai: 1 });

  assert.equal(formatKoordinat([101.6958, 2.9264]), '2.9264° N, 101.6958° E');

Day 5 · S3 — Best Practices

Quality & security checklist

Code

  • const by default · always === · no var
  • Small, pure functions, named with verbs
  • Every network await has try…catch + a loading state
  • npm run lint & npm test pass before committing
  • [lng, lat] in data, [lat, lng] only in Leaflet

Security

  • textContent / createElement — no innerHTML with external data
  • No secrets in the frontend, .env VITE_* or Git
  • Validate on the client and server (422)
  • No personal/classified data in browser storage
  • npm audit · lockfile committed · only trusted packages
